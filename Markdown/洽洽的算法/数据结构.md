## 并查集

**定义：**$每棵树为一个集合$

+ 合并操作：$合并两个元素所属集合$
+ 查询：$查询某个元素所属集合,判断两元素是否属于同一个集合$

**路径压缩：**$破坏了树的结构,将经过的元素直接连在根节点以加快查询,最坏时间复杂度O(m\log n)$

```cpp
struct DSU : vector<signed> {
    DSU(int n) : vector(n) { iota(begin(), end(), 0); }
    int find(int u) {
        for ( ; at(u) != u; u = at(u) = at(at(u)));
        return u;
    }
    int merge(int u, int v) { return find(u) == find(v) ? -1 : at(find(u)) = find(v); }
};
```

**启发式合并：**$不破坏原有结构,将秩小的并查集合并到秩大的并查集,最坏时间复杂度O(m\log n)$

```cpp
vector<int> fa(n + 1), r(n + 1, 1);
iota(fa.begin(), fa.end(), 0);	//初始化另一种写法
function<int(int)> find = [&](int u) -> int {
	return fa[u] == u ? u : find(fa[u]);
};
for (int i = 1, u, v; i <= m; i++) {
    cin >> u >> v;
    u = find(u), v = find(v);
    if (u == v) {
        continue;
    } else if (r[u] > r[v]) {
        swap(u, v);
    }
    fa[u] = v, r[v] += r[u];
}
```

**带权并查集：**

```cpp
vector<int> dep(n + 1);     // 维护深度，要保证根节点深度为 0 ，防止查询时重复计算贡献
auto find = [&](auto && find, int u) -> int {
    if (fa[u] == u) {
        return u;
    }
    int t = find(find, fa[u]);
    dep[u] += dep[fa[u]];
    return fa[u] = t;
};

auto merge = [&](int u, int v) {
    int fu = find(find, u), fv = find(find, v);
    if (fu != fv) {
        fa[fv] = u;
        dep[v] = 1;
        find(find, v);
    }
};
```

## ST表

**定义：**$ST表是用于解决可重复贡献问题的数据结构$

```cpp
int m = ceil(log2(n)) + 1;	//加一是为了开大空间防re
vector<int> a(n);
vector<vector<int>> st(n, vector<int>(m));
for (int i = 0; i < n; i++) {
    st[i][0] = a[i];
}
for (int j = 1; j < m; j++) {	//init
    int d = 1 << j - 1;	//判断是否越界
    for (int i = 0; i + d < n; i++) {
        st[i][j] = op(st[i][j - 1], st[i + (1 << j - 1)][j - 1]);
    }
}

int query(int l, int r) {
    if (l == r) {
		return a[l];
    }
    int t = r - l + 1;
    t = ceil(log2(t)) - 1;	//减一是提前减一
    return op(st[l][t], st[r - (1 << t) + 1][t]);
}
```

## 树状数组

**作用：**$支持单点修改和区间查询$

$将[1,n]拆分为不多于\log n段区间$

空间复杂度$O(n)$,单次查询或修改时间复杂度$O(\log n)$

```cpp
auto lowbit = [](int x) {
    return x & -x;
};

vector<int> tr(n + 1);

auto add = [&](int x, int k) {
    for ( ; x < tr.size(); x += lowbit(x)) {
        tr[x] += k;
    }
};

auto get = [&](int x, int res = 0) {
    for ( ; x; x -= lowbit(x)) {
        res += tr[x];
    }
    return res;
};
```

**区间修改、单点查询：**$维护差分数组$

**区间修改、区间查询：**$\Large 维护差分数组d_i和数组i\times d_i\Rightarrow \sum \limits_{i=l}^ra_i=((r+1)\times \sum \limits_{i=1}^{r}d_i-l\times \sum\limits_{i=1}^{l-1}d_i) - \sum\limits_{i=l}^r(i\times d_i)$

**固定一端的区间和的异或和：** 转化为形如 $ \bigoplus\limits_{i = 0}^n (p_n - p_i) $ ，对于第 $ bit $ 位，等价于查询 $ (p_n - p_i) \mod 2 ^ {bit + 1} \geqslant 2 ^ {bit} $ 的个数，然后分类讨论，分成 $ p_n - p_i < 0, 0 \leqslant p_n - p_i < 2 ^ {bit + 1} $ 和 $ p_n - p_i \geqslant 2 ^ {bit + 1} $ 三种情况查询即可

## 线段树

**作用：**$可在O(\log N)内实现单点修改,区间查询,区间修改(区间求和,求区间最大值,最小值)等$

$将每个长度不为1的区间划分成左右俩个区间递归求解,通过合并左右区间来得答案,也可进行区间操作$

最后的叶子节点即为原数组的信息，由于采用堆式存储，叶子节点为 $n$ 时，数组范围最大为 $2^{\lceil\log_2n\rceil+1}\leqslant 4n-5$

（若使用动态开点，则数组范围最大为 $2n-1$ ）

故空间复杂度为 $O(n)$

+ **区间加某个值与区间修改为某个值：**

```cpp
template <class Info, class Tag>
struct SegTree : vector<Info> {    // 左闭右开，注意初始化数组需要有 l, r
    using tr = vector<Info>;
    int n;
    vector<Tag> tag;

    void init(const tr & t) { // t.size() > 0
        n = t.size(), tr::assign(4 << __lg(t.size()), {}), tag.assign(tr::size(), {});
        auto self = [&](auto && self, int l, int r, int p = 0) {
            if (l + 1 == r) return tr::at(p) = t[l], void();
            int m = l + r >> 1;
            self(self, l, m, p << 1 | 1), self(self, m, r, p + 1 << 1), pull(p);
        };
        self(self, 0, t.size());
    }
    SegTree(int n) {
        tr t(max<int>(1, n));
        for (int i = 0; i < t.size(); t[i++] = Info(i, i + 1));
        init(t);
    }
    SegTree(const tr & t) { init(t); }

    void pull(int p) { tr::at(p).pull(tr::at(p << 1 | 1) + tr::at(p + 1 << 1)); }
    void update(int p, const Tag & t) {
        if (t != Tag()) tr::at(p).update(t), tag[p] += t;
    }
    void push(int p) {
        if (tag[p] != Tag()) update(p << 1 | 1, tag[p]), update(p + 1 << 1, tag[p]), tag[p] = {};
    }

    void update(int b, int e, const Tag & t) {
        auto self = [&](auto && self, int p, int l, int r) -> void {
            if (e <= l || r <= b) return;
            if (b <= l && r <= e) return update(p, t);
            int m = l + r >> 1;
            push(p), self(self, p << 1 | 1, l, m), self(self, p + 1 << 1, m, r), pull(p);
        };
        return self(self, 0, 0, n);
    }
    Info query(int b, int e) {
        auto self = [&](auto && self, int p, int l, int r) -> Info {
            if (e <= l || r <= b) return Info(b, e);
            if (b <= l && r <= e) return tr::at(p);
            int m = l + r >> 1;
            return push(p), self(self, p << 1 | 1, l, m) + self(self, p + 1 << 1, m, r);
        };
        return self(self, 0, 0, n);
    }
    template <class F>
    int FindFirst(int b, int e, F && chk) {
        auto self = [&](auto && self, int p, int l, int r) -> int {
            if (e <= l || r <= b || !chk(tr::at(p))) return -1;
            if (l + 1 == r) return l;
            int m = l + r >> 1, res = -1;
            return push(p), (res = self(self, p << 1 | 1, l, m)) == -1 ? self(self, p + 1 << 1, m, r) : res;
        };
        return self(self, 0, 0, n);
    }
    template <class F>
    int FindLast(int b, int e, F && chk) {
        auto self = [&](auto && self, int p, int l, int r) -> int {
            if (e <= l || r <= b || !chk(tr::at(p))) return -1;
            if (l + 1 == r) return l;
            int m = l + r >> 1, res = -1;
            return push(p), (res = self(self, p + 1 << 1, m, r)) == -1 ? self(self, p << 1 | 1, l, m) : res;
        };
        return self(self, 0, 0, n);
    }
};

template <class Info, class Tag>
struct DSegTree : vector<Info> {   // 多一个 chknode 可以开值域大小
    using tr = vector<Info>;
    int l, r;
    vector<array<signed, 2>> son;
    DSegTree(int l, int r) : l(l), r(r) { node(l, r); }
    int node(int l, int r, int p = -1) {
        if (p != -1) return p;
        return tr::emplace_back(l, r), son.emplace_back(array{-1, -1}), son.size() - 1;
    }
    void pull(int p) { tr::at(p).pull(tr::at(son[p][0]) + tr::at(son[p][1])); }

    void update(int b, int e, const Tag & t, int p = 0) {
        auto self = [&](auto && self, int p, int l, int r) -> int {
            if ((p = node(l, r, p)), e <= l || r <= b) return p;
            if (b <= l && r <= e) return tr::at(p).update(t), p;
            int m = l + r >> 1;
            son[p][0] = self(self, son[p][0], l, m), son[p][1] = self(self, son[p][1], m, r);
            return pull(p), p;
        };
        self(self, p, l, r);
    }
    Info query(int b, int e, int p) {
        auto self = [&](auto && self, int p, int l, int r) -> Info {
            if (p == -1 || e <= l || r <= b) return Info(l, r);
            if (b <= l && r <= e) return tr::at(p);
            int m = l + r >> 1;
            return self(self, son[p][0], l, m) + self(self, son[p][1], m, r);
        };
        return self(self, p, l, r);
    }
    template <class F>
    int FindFirst(int b, int e, F && chk, int p) {
        auto self = [&](auto && self, int p, int l, int r) -> int {
            if (p == -1 || e <= l || r <= b || !chk(tr::at(p))) return -1;
            if (l + 1 == r) return l;
            int m = l + r >> 1, res = -1;
            return (res = self(self, son[p][0], l, m)) == -1 ? self(self, son[p][1], m, r) : res;
        };
        return self(self, p, l, r);
    }
    template <class F>
    int FindLast(int b, int e, F && chk, int p) {
        auto self = [&](auto && self, int p, int l, int r) -> int {
            if (p == -1 || e <= l || r <= b || !chk(tr::at(p))) return -1;
            if (l + 1 == r) return l;
            int m = l + r >> 1, res = -1;
            return (res = self(self, son[p][1], m, r)) == -1 ? self(self, son[p][0], l, m) : res;
        };
        return self(self, p, l, r);
    }
    pair<int, int> merge(int t, int p) {    // tie(t, p) = merge(t, p), t 会清空为 -1
        auto self = [&](auto && self, int t, int p, int l, int r) -> int {
            if (t == -1 || p == -1) return node(l, r, max(t, p));
            if (l + 1 == r) return tr::at(p) += tr::at(t), p;
            int m = l + r >> 1;
            son[p][0] = self(self, son[t][0], son[p][0], l, m), son[p][1] = self(self, son[t][1], son[p][1], m, r);
            return pull(p), p;
        };
        return {-1, self(self, t, p, l, r)};
    }
    pair<int, int> split(int b, int e, int p) {    // tie(t, p) = split(b, e, p);
        auto self = [&](auto && self, int p, int l, int r, int t = -1) -> pair<int, int> {
            if ((t = node(l, r, t)), p == -1 || e <= l || r <= b) return {t, node(l, r, p)};
            if (b <= l && r <= e) return {p, t};
            int m = l + r >> 1;
            tie(son[t][0], son[p][0]) = self(self, son[p][0], l, m), tie(son[t][1], son[p][1]) = self(self, son[p][1], m, r);
            return pull(p), pull(t), pair{t, p};
        };
        return self(self, p, l, r);
    };
};

namespace mul_add {
    int mod;
    struct Info {
        unsigned l, r;
        int sum;
        Info(int l = -1, int r = -1, int sum = 0) : l(l), r(r), sum(sum) {}
        struct Tag {
            int mul = 1, add = 0;
            Tag & operator += (const Tag & t) {
                return (mul *= t.mul) %= mod, (add = add * t.mul + t.add) %= mod, *this;
            } Tag operator + (const Tag & t) const { return Tag(*this) += t; }
            bool & operator == (const Tag & t) const {
                return mul == t.mul && add == t.add;
            } bool & operator != (const Tag & t) const { return !(*this == t); }
        };
        Info & operator += (const Info & t) {
            return r = t.r, (sum += t.sum) %= mod, *this;
        } Info operator + (const Info & t) const { return Info(*this) += t; }
        void pull(const Info & t) { *this = t; }    // 是拉取的信息，一般等于即可
        void update(const Tag & t) { (sum = sum * t.mul + (r - l) * t.add) %= mod; }
    };
```

**权值线段树：**

一般需要离散化，维护左闭右闭的区间，一般相当于一个桶

查询最大的小于 $x$ 的数可以先用 $query$ 查询 $x$ 的排名 $rank$ ，然后查询全局第 $rank-cnt(x)$ 大的数即可

查询最小的大于 $x$ 的数同理

**主席树(可持久化线段树)：**

+ 查询区间答案 $ ans[l, r] = \begin{cases} ans[nxt_l + 1, r], & r \geqslant nxt_l \\ ans[l + 1, r], & otherwise \end{cases} $ ，令 $ ans[l] $ 是左端点为 $ l $ 的答案序列， $ ans[l] $ 可以由一部分的 $ ans[l + 1] $ 和另一部分的 $ ans[nxt_l + 1] $ 拼接而成，维护 $ ans[l + 1] $ 和 $ ans[nxt_l + 1] $ 的历史版本合并成新的版本

```cpp
struct node {
    int lson, rson, cnt;    //记录左儿子,右儿子,出现次数
};

vector<node> tr((n << 2) + (n * 18));   //空间为n * 4 + n log n

vector<int> root(n + 1);    //每个历史版本的根
int tot = 0;

vector<int> a(n + 1);

for (int i = 1; i <= n; i++) {
    cin >> a[i];
}

auto ta = a;

sort(ta.begin() + 1, ta.end());

ta.erase(unique(ta.begin() + 1, ta.end()), ta.end());	//离散化

auto get_val = [&](int x) {
    return lower_bound(ta.begin() + 1, ta.end(), x) - ta.begin();
};

auto pushup = [&](int p) {
    tr[p].cnt = tr[tr[p].lson].cnt + tr[tr[p].rson].cnt;
};

auto build = [&](auto build, int l, int r) -> int {
    int root = ++tot;   //新开节点
    if (l == r) return root;
    int mid = l + r >> 1;
    tr[root] = {build(build, l, mid), build(build, mid + 1, r)};
    return root;    //返回节点编号
};

auto insert = [&](auto insert, int l, int r, int k, int root) -> int {  //k为插入点的权值,dir为新节点
    int dir = ++tot;
    tr[dir] = tr[root]; //新开节点并继承
    if (l == r) {
        tr[dir].cnt++;
        return dir;
    }
    int mid = l + r >> 1;
    if (k <= mid) tr[dir].lson = insert(insert, l, mid, k, tr[root].lson);  //在左边向左边插
    else tr[dir].rson = insert(insert, mid + 1, r, k, tr[root].rson);   //在右边向右边插
    pushup(dir);    //向上传
    return dir; //返回节点编号
};

auto query = [&](auto query, int root, int dir, int l, int r, int k) -> int {   //root为l-1的节点,dir为r的节点
    if (l == r) return r;
    int cnt = tr[tr[dir].lson].cnt - tr[tr[root].lson].cnt; //前缀和思想
    int mid = l + r >> 1;
    if (k <= cnt) return query(query, tr[root].lson, tr[dir].lson, l, mid, k);  //在左边
    else return query(query, tr[root].rson, tr[dir].rson, mid + 1, r, k - cnt); //在右边,且k-cnt
};

root[0] = build(build, 1, ta.size() - 1);   //第一个历史版本

for (int i = 1; i <= n; i++) {
    root[i] = insert(insert,  1, ta.size() - 1, get_val(a[i]), root[i - 1]);
}

for (int i = 1; i <= m; i++) {
    int l, r, k;
    cin >> l >> r >> k;
    cout << ta[query(query, root[l - 1], root[r], 1, ta.size() - 1, k)] << "\n";
}
```

## 扫描线

**作用：**扫描面积、周长和二维数点

扫描线不需要 $pushdown$ 和 $query$ ，维护的为一段左闭右开的区间

**Atlantis:**

```cpp
vector<array<int, 4>> seg(1);
vector<int> X(1);

for (int i = 1; i <= n; i++) {
    int x1, y1, x2, y2;
    cin >> x1 >> y1 >> x2 >> y2;
    seg.push_back({x1, x2, y1, 1});     //加边
    seg.push_back({x1, x2, y2, -1});    //减边
    X.push_back(x1), X.push_back(x2);
}

sort(seg.begin() + 1, seg.end(), [&](auto x, auto y) {  //离散化
    return x[2] < y[2];    //当纵坐标相同时先计算加边再计算减边
});
sort(X.begin() + 1, X.end());
X.erase(unique(X.begin() + 1, X.end()), X.end());

struct node {
    int l, r;
    array<int, 2> LR = {0}; //扫描线维护的为线段
    int sum = 0, add = 0;	//需要赋初值，赋为幺元
};

vector<node> tr;

auto merge = [&](node l, node r, node u = {}) { //合并贡献
    u = {l.l, r.r, u.LR, l.sum + r.sum, u.add}; 	//记得补全
    return u;
};

auto pushup = [&](int p) {
    if (tr[p].add) {    //线段已经被覆盖
        tr[p].sum = tr[p].LR[1] - tr[p].LR[0];  
    } else if (tr[p].r == tr[p].l) {    //叶子节点，不从子节点得到答案，防止越界
        tr[p].sum = 0;
    } else {    //非叶子节点，从子节点获取答案
        tr[p] = merge(tr[p << 1], tr[p << 1 | 1], tr[p]);   //向上传
    }
};  

auto add_down = [&](int k, int p) {  //加法的更新
    tr[p].add += k;
    pushup(p);  //更新，扫描线的更新与pushup相同
};

auto build = [&](int sz) {  //建树
    tr = vector<node>(sz << 2);

    auto build = [&](auto build, int l, int r, int p = 1) {
        tr[p] = {l, r, {X[l], X[r + 1]}};
        if (l == r) {   //叶子节点等于原数组
            return;
        }
        int mid = l + r >> 1;
        build(build, l, mid, p << 1), build(build, mid + 1, r, p << 1 | 1);
        pushup(p);
    };

    build(build, 1, sz);
};

auto update = [&](auto update, int l, int r, int k, int p = 1) {    //更新
    if (l <= tr[p].LR[0] && tr[p].LR[1] <= r) {
        add_down(k, p); //添加标记,更新节点
        return;
    }
    if (l < tr[p << 1].LR[1]) update(update, l, r, k, p << 1); //左边有交集，因为为线段，相等时无交集
    if (tr[p << 1 | 1].LR[0] < r) update(update, l, r, k, p << 1 | 1); //右边有交集
    pushup(p);
};

build(X.size() - 2);    //线段树节点为区间，坐标最后一个点不能用

int ans = 0, y0 = seg[1][2];

for (int i = 1; i < seg.size(); i++) {
    auto [x1, x2, y, op] = seg[i];
    ans += tr[1].sum * (y - y0);
    update(update, x1, x2, op, 1);
    y0 = y;
}
```

```cpp
const int maxn = 5010 << 3;

struct {
    int l, r, len, cnt, lflag, rflag, num;		//不同点  num为不相交的线段数量, flag用来合并num
} tr[maxn];

void slove() {
    int n;
    cin >> n;
    vector<int> X;
    struct node {
        int y, x1, x2, op;
        bool operator< (const node& N) const {
            if (y == N.y) {
                return op > N.op;
            }
            return y < N.y;
        }
    };
    vector<node> seg;
    for (int x1, y1, x2, y2; n--; ) {
        cin >> x1 >> y1 >> x2 >> y2;
        X.push_back(x1), X.push_back(x2);
        seg.push_back({y1, x1, x2, 1}), seg.push_back({y2, x1, x2, -1});
    }
    sort(X.begin(), X.end());
    X.erase(unique(X.begin(), X.end()), X.end());
    sort(seg.begin(), seg.end());
    
    auto pushup = [&](int p) {		//不同点
        if (tr[p].cnt) {	//有覆盖
            tr[p].len = X[tr[p].r + 1] - X[tr[p].l];
            tr[p].rflag = tr[p].lflag = tr[p].num = 1;
        } else if (tr[p].l == tr[p].r) {	//叶子节点
            tr[p].len = tr[p].num = tr[p].lflag = tr[p].rflag = 0;
        } else {	
            tr[p].len = tr[p << 1].len + tr[p << 1 | 1].len;
            tr[p].num = tr[p << 1].num + tr[p << 1 | 1].num;
            tr[p].lflag = tr[p << 1].lflag, tr[p].rflag = tr[p << 1 | 1].rflag;
            if (tr[p << 1].rflag && tr[p << 1 | 1].lflag) tr[p].num--;		//重复计算
        }
    };

    function<void(int, int, int)> build = [&](int l, int r, int p) {
        tr[p] = {l, r, 0, 0};
        if (l == r) {
            return;
        }
        int mid = l + r >> 1;
        build(l, mid, p << 1), build(mid + 1, r, p << 1 | 1);
    };
    function<void(int, int, int, int)> update = [&](int l, int r, int p, int op) {
        if (l <= X[tr[p].l] && X[tr[p].r + 1] <= r) {
            tr[p].cnt += op;
            pushup(p);
            return;
        }
        int mid = tr[p].l + tr[p].r >> 1;
        if (l < X[mid + 1]) update(l, r, p << 1, op);
        if (X[mid + 1] < r) update(l, r, p << 1 | 1, op);
        pushup(p);
    };
    build(0, X.size() - 2, 1);
    int ans = 0, y0 = seg[0].y, len0 = 0, op0 = 1;
    for (auto [y, x1, x2, op] : seg) {		//一定要先计算完"加边"再计算"减边",且计算完"加边"要再计算一次
        if (y == y0) {	//同一高度
            if (op0 == op) {	//都为加边或都为减边
                update(x1, x2, 1, op);
                continue;
            } else {	//加边算完计算一次
                ans += abs(tr[1].len - len0);
                len0 = tr[1].len, op0 = -1;	//变减边
                update(x1, x2, 1, op);
            }
        } else {
            op0 = 1;	//不同高度从加边开始,顺便计算一次
            ans += abs(tr[1].len - len0) + tr[1].num * (y - y0) * 2;
            len0 = tr[1].len, y0 = y;
            update(x1, x2, 1, op);
        }
    }
    ans += len0;	//加上没被计算的
    cout << ans << "\n";
}
```
